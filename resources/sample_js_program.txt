// Import the Express module
const express = require('express');

// Initialize the Express application
const app = express();

// Define the port number
const PORT = 3000;

// Built-in middleware to parse incoming JSON request bodies
app.use(express.json());

// 1. Basic GET Route (Home Page)
app.get('/', (req, res) => {
    res.send('Welcome to your first Express.js Server!');
});

// 2. GET Route with a Route Parameter
app.get('/api/users/:id', (req, res) => {
    const userId = req.params.id;
    res.json({
        message: `Fetched data for user ID: ${userId}`,
        status: 'success'
    });
});

// 3. POST Route to receive data
app.post('/api/users', (req, res) => {
    const newUserData = req.body;
    res.status(201).json({
        message: 'User created successfully!',
        data: newUserData
    });
});

// 4. Wildcard Route to handle 404 Errors (Page Not Found)
app.use((req, res) => {
    res.status(404).send('Resource not found.');
});

// Start the server and listen for connections
app.listen(PORT, () => {
    console.log(`Server is running at http://localhost:${PORT}`);
});
